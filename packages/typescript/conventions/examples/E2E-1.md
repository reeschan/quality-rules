## 悪い例

```ts
// 割引率の場合分けを、画面を動かす E2E で 1 つずつ確かめている。
for (const [quantity, expectedRate] of discountCases) {
  test(`数量 ${quantity} のときの割引率`, async ({ page }) => {
    await page.goto('/cart');
    await page.getByLabel('数量').fill(String(quantity));
    await expect(page.getByTestId('rate')).toHaveText(expectedRate);
  });
}
```

## 良い例

```ts
// 場合分けは、速い単体テストで確かめる。
test.each(discountCases)('数量 %i のときの割引率は %s', (quantity, expectedRate) => {
  expect(calcDiscountRate(quantity)).toBe(expectedRate);
});

// E2E は、購入までの代表の流れが 1 本通ることだけを確かめる。
test('カートに入れて購入を完了できる', async ({ page }) => {
  await page.goto('/cart');
  await page.getByRole('button', { name: '購入する' }).click();
  await expect(page.getByText('ご注文ありがとうございました')).toBeVisible();
});
```
