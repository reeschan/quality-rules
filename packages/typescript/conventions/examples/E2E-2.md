## 悪い例

```ts
test('メモを追加できる', async ({ page }) => {
  await page.goto('/notes');
  await page.locator('.note-form > div:nth-child(2) > button').click();
  // 内部の状態を直接読んでいる。
  const state = await page.evaluate(() => window.__store.getState().notes.length);
  expect(state).toBe(1);
});
```

## 良い例

```ts
test('メモを追加できる', async ({ page }) => {
  await page.goto('/notes');
  await page.getByRole('textbox', { name: 'メモ' }).fill('牛乳を買う');
  await page.getByRole('button', { name: '追加' }).click();
  await expect(page.getByRole('listitem')).toHaveText('牛乳を買う');
});
```
